```rust
#![no_std]
#![no_main]

use aya_ebpf::{
    macros::{map, tracepoint},
    maps::RingBuf,
    programs::TracePointContext,
    helpers::bpf_get_current_pid_tgid,
    helpers::bpf_get_current_comm,
    EbpfContext,
};
use aya_log_ebpf::info;

/// Event pushed to userspace via ring buffer.
/// Événement poussé vers l'espace utilisateur via le ring buffer.
#[repr(C)]
pub struct SocketEvent {
    pub pid: u32,
    pub tgid: u32,
    pub comm: [u8; 16],
    pub sport: u16,
    pub dport: u16,
    pub family: u16, // AF_INET=2, AF_INET6=10
    pub protocol: u8, // IPPROTO_TCP=6, IPPROTO_UDP=17
    pub oldstate: u32,
    pub newstate: u32,
}

#[map]
static EVENTS: RingBuf = RingBuf::with_byte_size(256 * 1024, 0);

/// Tracepoint: sock/inet_sock_set_state
/// Fires on every TCP state transition (SYN_SENT, ESTABLISHED, CLOSE, etc.)
///
/// Point de trace : sock/inet_sock_set_state
/// Se déclenche à chaque transition d'état TCP
#[tracepoint]
pub fn inet_sock_set_state(ctx: TracePointContext) -> u32 {
    match try_inet_sock_set_state(&ctx) {
        Ok(0) => 0,
        Ok(ret) => ret,
        Err(_) => 1,
    }
}

fn try_inet_sock_set_state(ctx: &TracePointContext) -> Result<u32, i64> {
    // Read tracepoint fields from context
    // Format: struct trace_event_raw_inet_sock_set_state
    // Offsets: oldstate(8), newstate(12), sport(14), dport(16), family(18), protocol(20)
    let oldstate: u32 = unsafe { ctx.read_at(8)? };
    let newstate: u32 = unsafe { ctx.read_at(12)? };
    let sport: u16 = unsafe { ctx.read_at(14)? };
    let dport: u16 = unsafe { ctx.read_at(16)? };
    let family: u16 = unsafe { ctx.read_at(18)? };
    let protocol: u8 = unsafe { ctx.read_at(20)? };

    // Only capture IPv4 and IPv6
    if family != 2 && family != 10 {
        return Ok(0);
    }

    let pid_tgid = bpf_get_current_pid_tgid();
    let pid = (pid_tgid >> 32) as u32;
    let tgid = pid_tgid as u32;

    let mut comm = [0u8; 16];
    let _ = bpf_get_current_comm(&mut comm);

    // Reserve space in ring buffer and write event
    if let Some(mut buf) = EVENTS.reserve::<SocketEvent>(0) {
        let event = buf.as_mut_ptr();
        unsafe {
            (*event).pid = pid;
            (*event).tgid = tgid;
            (*event).comm = comm;
            (*event).sport = sport;
            (*event).dport = dport;
            (*event).family = family;
            (*event).protocol = protocol;
            (*event).oldstate = oldstate;
            (*event).newstate = newstate;
        }
        buf.submit(0);
    }

    Ok(0)
}

#[panic_handler]
fn panic(_info: &core::panic::PanicInfo) -> ! {
    unsafe { core::hint::unreachable_unchecked() }
}
```

```rust
pub mod events;

use std::net::IpAddr;
use std::sync::Arc;

use async_trait::async_trait;
use dashmap::DashMap;
use tracing::{debug, info, warn, error};

use syswall_domain::entities::ProcessInfo;
use syswall_domain::errors::DomainError;
use syswall_domain::ports::ProcessResolver;
use syswall_domain::value_objects::{ExecutablePath, Protocol};

use events::SocketEvent;

/// Maps (protocol, local_port) → PID for fast lookups.
/// Associe (protocole, port_local) → PID pour des recherches rapides.
type SocketMap = DashMap<(u8, u16), u32>;

/// eBPF-based process resolver. Loads a BPF program that hooks into
/// inet_sock_set_state tracepoint and captures PID per socket.
///
/// Résolveur de processus basé sur eBPF.
pub struct EbpfProcessResolver {
    socket_map: Arc<SocketMap>,
    _ebpf_handle: tokio::task::JoinHandle<()>,
}

impl EbpfProcessResolver {
    /// Attempt to load the eBPF program. Returns Err if loading fails.
    /// Tente de charger le programme eBPF. Retourne Err en cas d'échec.
    pub fn try_new() -> Result<Self, DomainError> {
        let bpf_bytes = include_bytes_aligned!(
            concat!(env!("OUT_DIR"), "/syswall-ebpf-prog")
        );

        let mut ebpf = aya::Ebpf::load(bpf_bytes).map_err(|e| {
            DomainError::Infrastructure(format!("Failed to load eBPF program: {}", e))
        })?;

        // Attach to tracepoint
        use aya::programs::TracePoint;
        let program: &mut TracePoint = ebpf
            .program_mut("inet_sock_set_state")
            .ok_or_else(|| DomainError::Infrastructure("BPF program not found".into()))?
            .try_into()
            .map_err(|e| DomainError::Infrastructure(format!("BPF program type error: {}", e)))?;

        program.load().map_err(|e| {
            DomainError::Infrastructure(format!("Failed to load BPF tracepoint: {}", e))
        })?;
        program.attach("sock", "inet_sock_set_state").map_err(|e| {
            DomainError::Infrastructure(format!("Failed to attach BPF tracepoint: {}", e))
        })?;

        info!("eBPF tracepoint inet_sock_set_state attached successfully");

        // Setup ring buffer reader
        let socket_map: Arc<SocketMap> = Arc::new(DashMap::new());
        let map_clone = socket_map.clone();

        use aya::maps::RingBuf;
        let ring_buf = RingBuf::try_from(
            ebpf.take_map("EVENTS")
                .ok_or_else(|| DomainError::Infrastructure("EVENTS map not found".into()))?,
        )
        .map_err(|e| DomainError::Infrastructure(format!("RingBuf error: {}", e)))?;

        let handle = tokio::spawn(async move {
            Self::drain_ring_buffer(ring_buf, map_clone).await;
        });

        Ok(Self {
            socket_map,
            _ebpf_handle: handle,
        })
    }

    /// Continuously drain events from the ring buffer and update the socket map.
    /// Draine continuellement les événements du ring buffer et met à jour la table de sockets.
    async fn drain_ring_buffer(mut ring_buf: aya::maps::RingBuf<aya::maps::MapData>, map: Arc<SocketMap>) {
        loop {
            while let Some(event_data) = ring_buf.next() {
                if event_data.len() < std::mem::size_of::<SocketEvent>() {
                    continue;
                }
                let event: SocketEvent = unsafe {
                    std::ptr::read_unaligned(event_data.as_ptr() as *const SocketEvent)
                };

                // Store the mapping: (protocol, sport) → pid
                if event.pid > 0 && event.sport > 0 {
                    map.insert((event.protocol, event.sport), event.pid);
                    debug!(
                        "eBPF: {}:{} pid={} ({})",
                        event.protocol, event.sport, event.pid, event.comm_str()
                    );
                }
            }
            // Yield to avoid busy-looping
            tokio::time::sleep(std::time::Duration::from_millis(10)).await;
        }
    }

    fn protocol_to_u8(protocol: Protocol) -> u8 {
        match protocol {
            Protocol::Tcp => 6,  // IPPROTO_TCP
            Protocol::Udp => 17, // IPPROTO_UDP
            _ => 0,
        }
    }
}

#[async_trait]
impl ProcessResolver for EbpfProcessResolver {
    async fn resolve(&self, pid: u32) -> Result<Option<ProcessInfo>, DomainError> {
        // Read /proc/<pid>/exe and /proc/<pid>/status
        let result = tokio::task::spawn_blocking(move || {
            read_process_info_from_proc(pid)
        })
        .await
        .map_err(|e| DomainError::Infrastructure(format!("spawn_blocking: {}", e)))?;
        Ok(result)
    }

    async fn resolve_by_socket(&self, _inode: u64) -> Result<Option<ProcessInfo>, DomainError> {
        Ok(None) // Not applicable for eBPF resolver
    }

    async fn resolve_by_connection(
        &self,
        protocol: Protocol,
        _local_ip: IpAddr,
        local_port: u16,
        _remote_ip: IpAddr,
        _remote_port: u16,
    ) -> Result<Option<ProcessInfo>, DomainError> {
        let proto = Self::protocol_to_u8(protocol);
        if let Some(entry) = self.socket_map.get(&(proto, local_port)) {
            let pid = *entry;
            return self.resolve(pid).await;
        }
        Ok(None)
    }
}

/// Combines eBPF (fast, kernel-level) with procfs fallback.
/// Combine eBPF (rapide, niveau kernel) avec un fallback procfs.
pub struct HybridProcessResolver {
    ebpf: Option<EbpfProcessResolver>,
    fallback: Arc<dyn ProcessResolver>,
}

impl HybridProcessResolver {
    pub fn new(ebpf: Option<EbpfProcessResolver>, fallback: Arc<dyn ProcessResolver>) -> Self {
        Self { ebpf, fallback }
    }
}

#[async_trait]
impl ProcessResolver for HybridProcessResolver {
    async fn resolve(&self, pid: u32) -> Result<Option<ProcessInfo>, DomainError> {
        if let Some(ref ebpf) = self.ebpf {
            if let Ok(Some(info)) = ebpf.resolve(pid).await {
                return Ok(Some(info));
            }
        }
        self.fallback.resolve(pid).await
    }

    async fn resolve_by_socket(&self, inode: u64) -> Result<Option<ProcessInfo>, DomainError> {
        self.fallback.resolve_by_socket(inode).await
    }

    async fn resolve_by_connection(
        &self,
        protocol: Protocol,
        local_ip: IpAddr,
        local_port: u16,
        remote_ip: IpAddr,
        remote_port: u16,
    ) -> Result<Option<ProcessInfo>, DomainError> {
        if let Some(ref ebpf) = self.ebpf {
            if let Ok(Some(info)) = ebpf.resolve_by_connection(
                protocol, local_ip, local_port, remote_ip, remote_port
            ).await {
                return Ok(Some(info));
            }
        }
        self.fallback.resolve_by_connection(
            protocol, local_ip, local_port, remote_ip, remote_port
        ).await
    }
}

/// Read process info from /proc/<pid>. Shared helper used by both resolvers.
/// Lit les informations du processus depuis /proc/<pid>.
fn read_process_info_from_proc(pid: u32) -> Option<ProcessInfo> {
    let proc_path = std::path::PathBuf::from(format!("/proc/{}", pid));
    if !proc_path.exists() {
        return None;
    }

    let exe_path = std::fs::read_link(proc_path.join("exe"))
        .ok()
        .and_then(|p| {
            let s = p.to_string_lossy().to_string();
            let clean = if s.ends_with(" (deleted)") {
                std::path::PathBuf::from(&s[..s.len() - 10])
            } else {
                p
            };
            ExecutablePath::new(clean).ok()
        });

    let cmdline = std::fs::read(proc_path.join("cmdline"))
        .ok()
        .and_then(|b| {
            let s = String::from_utf8_lossy(&b).replace('\0', " ").trim().to_string();
            if s.is_empty() { None } else { Some(s) }
        });

    let status = std::fs::read_to_string(proc_path.join("status")).ok()?;
    let name = status.lines()
        .find(|l| l.starts_with("Name:"))
        .map(|l| l.trim_start_matches("Name:").trim().to_string())?;

    Some(ProcessInfo {
        pid,
        name,
        path: exe_path,
        cmdline,
        icon: None, // Resolved by IconResolver in the daemon layer
    })
}
```

```typescript
import { writable } from 'svelte/store';

export interface ToastMessage {
  id: string;
  message: string;
  variant: 'success' | 'error' | 'warning' | 'info';
  duration?: number;
}

const { subscribe, update } = writable<ToastMessage[]>([]);

export const toasts = { subscribe };

export function addToast(message: string, variant: ToastMessage['variant'] = 'info', duration = 4000) {
  const id = crypto.randomUUID();
  update(all => [...all, { id, message, variant, duration }]);
  if (duration > 0) {
    setTimeout(() => removeToast(id), duration);
  }
}

export function removeToast(id: string) {
  update(all => all.filter(t => t.id !== id));
}
```

```bash
post_install() {
    # Create syswall group for socket access
    if ! getent group syswall > /dev/null 2>&1; then
        groupadd syswall
    fi

    # Create directories
    mkdir -p /var/lib/syswall /var/log/syswall /var/run/syswall

    # Set capabilities on daemon binary
    setcap 'cap_net_admin,cap_net_raw,cap_sys_ptrace,cap_dac_read_search,cap_bpf,cap_perfmon=ep' /usr/bin/syswall-daemon 2>/dev/null || true

    # Reload systemd
    systemctl daemon-reload

    echo ">>> Add your user to the syswall group: sudo usermod -aG syswall \$USER"
    echo ">>> Then start: sudo systemctl start syswall"
}

post_upgrade() {
    post_install

    # Restart if running
    if systemctl is-active --quiet syswall; then
        systemctl restart syswall
    fi
}

pre_remove() {
    systemctl stop syswall 2>/dev/null || true
    systemctl disable syswall 2>/dev/null || true
}

post_remove() {
    systemctl daemon-reload
    echo ">>> Config preserved in /etc/syswall/. Remove manually if desired."
}
```

- [ ] **Step 2: Write PolicyEngine benchmark**

`crates/app/benches/policy_bench.rs`:
```rust
use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
use chrono::Utc;
use syswall_domain::entities::*;
use syswall_domain::events::DefaultPolicy;
use syswall_domain::services::PolicyEngine;
use syswall_domain::value_objects::*;

fn make_connection() -> Connection {
    Connection {
        id: ConnectionId::new(),
        protocol: Protocol::Tcp,
        source: SocketAddress::new("192.168.1.100".parse().unwrap(), Port::new(45000).unwrap()),
        destination: SocketAddress::new("93.184.216.34".parse().unwrap(), Port::new(443).unwrap()),
        direction: Direction::Outbound,
        state: ConnectionState::New,
        process: Some(ProcessInfo {
            pid: 1234,
            name: "firefox".to_string(),
            path: None,
            cmdline: None,
            icon: None,
        }),
        user: Some(SystemUser { uid: 1000, name: "seb".to_string() }),
        bytes_sent: 0,
        bytes_received: 0,
        started_at: Utc::now(),
        verdict: ConnectionVerdict::Unknown,
        matched_rule: None,
        remote_hostname: None,
    }
}

fn make_rules(count: usize) -> Vec<Rule> {
    (0..count).map(|i| {
        Rule {
            id: RuleId::new(),
            name: format!("Rule {}", i),
            priority: RulePriority::new(i as u32).unwrap(),
            enabled: true,
            criteria: RuleCriteria {
                remote_port: Some(PortMatcher::Exact(Port::new((8000 + i) as u16).unwrap())),
                protocol: Some(Protocol::Tcp),
                ..Default::default()
            },
            effect: if i % 2 == 0 { RuleEffect::Allow } else { RuleEffect::Block },
            scope: RuleScope::Permanent,
            source: RuleSource::Manual,
            created_at: Utc::now(),
            updated_at: Utc::now(),
        }
    }).collect()
}

fn policy_engine_benchmark(c: &mut Criterion) {
    let conn = make_connection();
    let mut group = c.benchmark_group("PolicyEngine::evaluate");

    for rule_count in [10, 100, 500, 1000] {
        let rules = make_rules(rule_count);
        group.bench_with_input(
            BenchmarkId::new("rules", rule_count),
            &rules,
            |b, rules| {
                b.iter(|| {
                    PolicyEngine::evaluate(black_box(&conn), black_box(rules), DefaultPolicy::Block)
                });
            },
        );
    }

    group.finish();
}

criterion_group!(benches, policy_engine_benchmark);
criterion_main!(benches);
```
